// ----- 方法2：使用 HttpURLConnection（更底层，能看到更多信息）-----
function fetchWithHttpURLConnection(urlStr) {
    logi("📥 [方法2] 使用 HttpURLConnection...");

    try {
        var url = new URL(urlStr);
        var connection = url.openConnection();

        // 设置请求头
        connection.setRequestProperty("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36");
        connection.setRequestProperty("Referer", "https://v.smtcdns.com/");
        connection.setRequestProperty("Accept", "*/*");
        connection.setRequestProperty("Accept-Language", "zh-CN,zh;q=0.9,en;q=0.8");
        connection.setRequestProperty("Connection", "keep-alive");
        connection.setConnectTimeout(CONFIG.timeout);
        connection.setReadTimeout(CONFIG.timeout);
        connection.setInstanceFollowRedirects(true); // 自动跟随重定向

        connection.connect();

        var statusCode = connection.getResponseCode();
        logi("📊 状态码: " + statusCode);
        logi("📊 响应消息: " + connection.getResponseMessage());

        // 打印所有响应头（调试用）
        var headerFields = connection.getHeaderFields();
        var headerKeys = headerFields.keySet().iterator();
        logi("📊 响应头:");
        while (headerKeys.hasNext()) {
            var key = headerKeys.next();
            var value = headerFields.get(key);
            if (key) {
                logi("   " + key + ": " + value);
            }
        }

        if (statusCode != 200) {
            // 尝试读取错误流
            try {
                var errorStream = connection.getErrorStream();
                if (errorStream) {
                    var reader = new BufferedReader(new InputStreamReader(errorStream));
                    var errorMsg = "";
                    var line = "";
                    while ((line = reader.readLine()) != null) {
                        errorMsg += line;
                    }
                    reader.close();
                    logi("❌ 错误信息: " + errorMsg);
                }
            } catch (e) {}
            connection.disconnect();
            return null;
        }

        // 读取内容
        var inputStream = connection.getInputStream();
        var reader = new BufferedReader(new InputStreamReader(inputStream, "UTF-8"));
        var content = "";
        var line = "";
        while ((line = reader.readLine()) != null) {
            content += line + "\n";
        }
        reader.close();
        inputStream.close();
        connection.disconnect();

        logi("✅ 获取成功，内容长度: " + content.length);
        if (content.length > 0) {
            logi("📄 内容预览: " + content.substring(0, Math.min(300, content.length)));
        } else {
            logi("⚠️ 内容为空");
        }
        return content;

    } catch (e) {
        logi("❌ HttpURLConnection 异常: " + e.message);
        e.printStackTrace();
        return null;
    }
}

// 清理临时文件
function cleanTemp(folderPath) {
    try {
        var folder = new File(folderPath);
        if (folder.exists()) {
            var files = folder.listFiles();
            if (files) {
                for (var i = 0; i < files.length; i++) {
                    try {
                        files[i].delete();
                    } catch (e) {}
                }
            }
            folder.delete();
            logi("🧹 临时文件已清理");
        }
    } catch (e) {
        logi("⚠️ 清理失败: " + e.message);
    }
}

// ----- 解析 M3U8 -----
function parseM3U8(m3u8Content, m3u8Url) {
    logi("🔍 开始解析 M3U8...");

    if (!m3u8Content || m3u8Content.length == 0) {
        logi("❌ 内容为空");
        return null;
    }

    var lines = m3u8Content.split("\n");
    var tsUrls = [];

    // 获取基础 URL（去掉最后的文件名）
    var baseUrl = m3u8Url.substring(0, m3u8Url.lastIndexOf("/") + 1);
    logi("🏗️ 基础 URL: " + baseUrl);

    for (var i = 0; i < lines.length; i++) {
        var line = lines[i].trim();
        if (line == "" || line.startsWith("#")) {
            continue;
        }

        // 这里 line 是相对路径，需要拼接完整 URL
        // 例如: 00_gzc_...ts?index=0&...
        var fullUrl = baseUrl + line;
        tsUrls.push(fullUrl);

        if (tsUrls.length <= 3) {
            logi("🔗 示例 URL " + tsUrls.length + ": " + fullUrl);
        }
    }

    logi("✅ 解析完成，共 " + tsUrls.length + " 个片段");
    return tsUrls;
}

// 直接拼接 TS 文件
function mergeTsFiles(tempFolder, outputPath) {
    logi("🔗 开始合并 TS 片段...");

    var folder = new File(tempFolder);
    if (!folder.exists()) {
        logi("❌ 临时文件夹不存在");
        return false;
    }

    var files = folder.listFiles();
    if (!files || files.length == 0) {
        logi("❌ 临时文件夹为空");
        return false;
    }

    // 按文件名排序
    var tsFiles = [];
    for (var i = 0; i < files.length; i++) {
        var fileName = files[i].getName();
        if (fileName.endsWith(".ts") && files[i].length() > 0) {
            tsFiles.push(files[i]);
        }
    }
    tsFiles.sort(function(a, b) {
        return a.getName().localeCompare(b.getName());
    });

    if (tsFiles.length == 0) {
        logi("❌ 没有找到有效的 TS 文件");
        return false;
    }

    logi("📦 找到 " + tsFiles.length + " 个有效 TS 文件");

    try {
        var outputFile = new File(outputPath);
        // 如果输出文件已存在，先删除
        if (outputFile.exists()) {
            outputFile.delete();
        }
        outputFile.getParentFile().mkdirs();

        var outputStream = new BufferedOutputStream(new FileOutputStream(outputFile));
        var buffer = java.lang.reflect.Array.newInstance(java.lang.Byte.TYPE, 65536);

        var totalSize = 0;
        for (var i = 0; i < tsFiles.length; i++) {
            var inputStream = new BufferedInputStream(new java.io.FileInputStream(tsFiles[i]));
            var bytesRead = 0;
            while ((bytesRead = inputStream.read(buffer)) != -1) {
                outputStream.write(buffer, 0, bytesRead);
                totalSize += bytesRead;
            }
            inputStream.close();

            if ((i + 1) % 5 == 0 || (i + 1) == tsFiles.length) {
                logi("⏳ 合并进度: " + (i + 1) + "/" + tsFiles.length);
            }
        }

        outputStream.flush();
        outputStream.close();

        var sizeMB = (totalSize / (1024 * 1024)).toFixed(2);
        logi("✅ 合并完成！文件大小: " + sizeMB + " MB");
        logi("📁 保存路径: " + outputPath);
        return true;

    } catch (e) {
        logi("❌ 合并失败: " + e.message);
        e.printStackTrace();
        return false;
    }
}
